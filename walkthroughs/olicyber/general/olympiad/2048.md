# URL
https://training.olicyber.it/challenges#challenge-31
# Concept
* sending responses to a remote server in a programmatic way
# Method of solve
* we have to solve many many (2048) math equations sent from the server to our client
* we can use Pwntools to script out the response:
```Python
from pwn import *

op_list = ["DIFFERENZA","DIVISIONE_INTERA","SOMMA","PRODOTTO","POTENZA"]

p = remote("2048.challs.olicyber.it", 10007)

def parse_response(response):
    response = response.decode('UTF-8')
    operation = ""
    first_num = 0
    second_num = 0
    if response.split()[0] in op_list:
        first_num = int(response.split()[1])
        second_num = int(response.split()[2])
        if response.split()[0] == "DIFFERENZA":
            operation = "subtraction"
        if response.split()[0] == "DIVISIONE_INTERA":
            operation = "int_division"
        if response.split()[0] == "SOMMA":
            operation = "addition"
        if response.split()[0] == "PRODOTTO":
            operation = "multiplication"
        if response.split()[0] == "POTENZA":
            operation = "exponentiation"
    return operation,first_num,second_num

def do_the_math(operation,first_num,second_num):
    result = 0
    if operation == "subtraction":
        result = first_num - second_num
    if operation == "int_division":
        result = first_num // second_num
    if operation == "addition":
        result = first_num + second_num
    if operation == "multiplication":
        result = first_num * second_num
    if operation == "exponentiation":
        result = first_num ** second_num
    return result

first_response = p.recvline()
second_response = p.recvline()

while True:
    response = p.recv(1024)
    print(f"Received line: {response}")
    print(f"Length of response: {len(response.split())}")
    if len(response.split()) == 3:
        operation,first_num,second_num = parse_response(response)
        res_raw = do_the_math(operation,first_num,second_num)
        res_formatted = str(res_raw).encode()
        print(f"Sending answer: {res_formatted}")
        p.sendline(res_formatted)
```
